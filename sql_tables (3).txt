CREATE TABLE "BusinessAdmin" (
  business_admin_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  name TEXT NOT NULL,
  email TEXT UNIQUE NOT NULL,
  phone TEXT UNIQUE NOT NULL,
  firebase_id TEXT,              -- Nullable (can be empty initially)
  created_at TIMESTAMPTZ DEFAULT NOW()
);




CREATE TABLE "School" (
  school_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  school_name TEXT NOT NULL,
  email TEXT UNIQUE NOT NULL,
  phone TEXT UNIQUE NOT NULL,
  tagline TEXT,
  address TEXT,
  is_active BOOLEAN DEFAULT true, -- Renamed from isactive
  firebase_id TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);




CREATE TABLE "School_Status_Log" (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  school_id BIGINT NOT NULL REFERENCES "School"(school_id) ON DELETE RESTRICT,
  action_type TEXT NOT NULL CHECK (action_type IN ('SUSPEND', 'ACTIVATE')), 
  reason TEXT NOT NULL,       -- Now Mandatory
  changed_by BIGINT NOT NULL, -- Now Mandatory (ID of Business Admin)
  created_at TIMESTAMPTZ DEFAULT NOW()
);




CREATE TABLE "AcademicSession" (
  session_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  session_name TEXT UNIQUE NOT NULL,
  start_date DATE NOT NULL,
  end_date DATE NOT NULL,
  is_active BOOLEAN DEFAULT false, -- Defaults to false
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Enforce only ONE active session at a time
CREATE UNIQUE INDEX idx_single_active_session
ON "AcademicSession" (is_active)
WHERE is_active = true;




CREATE TABLE "Student" (
  student_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  -- school_id REMOVED: Students now tracked via Student_School_Enrollment
  student_name TEXT NOT NULL,
  parent_name TEXT NOT NULL,
  email TEXT UNIQUE,              -- Optional but unique if present
  phone TEXT NOT NULL,            -- NOT unique (siblings can share parent phone)
  is_active BOOLEAN DEFAULT true, -- Global student status (not school-specific)
  firebase_id TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Index for fast phone lookups (e.g. "all kids with phone 999")
CREATE INDEX idx_student_phone ON "Student"(phone);









-- Tracks student enrollment history across schools
CREATE TABLE "Student_School_Enrollment" (
  enrollment_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

  student_id BIGINT NOT NULL
    REFERENCES "Student"(student_id) ON DELETE RESTRICT,

  school_id BIGINT NOT NULL
    REFERENCES "School"(school_id) ON DELETE RESTRICT,

  is_active BOOLEAN NOT NULL DEFAULT TRUE,  -- Only one TRUE per student

  enrolled_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  left_school_at TIMESTAMPTZ DEFAULT NULL,
  leaving_reason TEXT DEFAULT NULL,

  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),

  -- Optional but recommended consistency rule
  CONSTRAINT chk_active_dates
  CHECK (
    (is_active = TRUE AND left_school_at IS NULL)
    OR
    (is_active = FALSE AND left_school_at IS NOT NULL)
  )
);

-- Performance indexes
CREATE INDEX idx_student_school_enrollment_student
ON "Student_School_Enrollment"(student_id);

CREATE INDEX idx_student_school_enrollment_school
ON "Student_School_Enrollment"(school_id);

-- 🚨 KEY RULE: only ONE active enrollment per student
CREATE UNIQUE INDEX idx_unique_student_active
ON "Student_School_Enrollment"(student_id)
WHERE is_active = TRUE;

-- Optional: prevent duplicate active rows for same school
CREATE UNIQUE INDEX idx_unique_student_school_active
ON "Student_School_Enrollment"(student_id, school_id)
WHERE is_active = TRUE;
















CREATE TABLE "Subject" (
  subject_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  school_id BIGINT NOT NULL REFERENCES "School"(school_id) ON DELETE RESTRICT,
  session_id BIGINT NOT NULL REFERENCES "AcademicSession"(session_id) ON DELETE RESTRICT,
  subject_name TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  
  -- Prevent duplicate subjects for the same school in the same session
  UNIQUE (school_id, session_id, subject_name)
);




CREATE TABLE "Class" (
  class_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  school_id BIGINT NOT NULL REFERENCES "School"(school_id) ON DELETE RESTRICT,
  session_id BIGINT NOT NULL REFERENCES "AcademicSession"(session_id) ON DELETE RESTRICT,
  class_name TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW(),

  -- A school cannot have two classes with the same name in the same session
  UNIQUE (school_id, session_id, class_name)
);




CREATE TABLE "Class_Subject_Relation" (
  class_subject_relation_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  class_id BIGINT NOT NULL REFERENCES "Class"(class_id) ON DELETE RESTRICT,
  subject_id BIGINT NOT NULL REFERENCES "Subject"(subject_id) ON DELETE RESTRICT,
  created_at TIMESTAMPTZ DEFAULT NOW(),

  -- Prevent assigning the same subject to the same class twice
  UNIQUE (class_id, subject_id)
);




CREATE TABLE "Student_Class_Enrollment_Relation" (
  enrollment_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  student_id BIGINT NOT NULL REFERENCES "Student"(student_id) ON DELETE RESTRICT,
  class_id BIGINT NOT NULL REFERENCES "Class"(class_id) ON DELETE RESTRICT,
  roll_no INTEGER, 
  created_at TIMESTAMPTZ DEFAULT NOW(),

  -- A student can only be in ONE class at a time
  UNIQUE (student_id), 

  -- No two students can have the same Roll No in the same class
  UNIQUE (class_id, roll_no)
);




CREATE TABLE "Teacher" (
  teacher_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  -- school_id REMOVED: Now tracked via Teacher_School_Enrollment
  name TEXT NOT NULL,

  -- Login / Contact Info
  phone TEXT NOT NULL,
  email TEXT,
  firebase_id TEXT,

  -- Status Flag (True = Active, False = Resigned/Suspended)
  is_active BOOLEAN DEFAULT TRUE,

  created_at TIMESTAMPTZ DEFAULT NOW(),

  UNIQUE (phone),
  UNIQUE (email)
);


-- Tracks teacher employment history across schools
CREATE TABLE "Teacher_School_Enrollment" (
  enrollment_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  
  teacher_id BIGINT NOT NULL REFERENCES "Teacher"(teacher_id) ON DELETE RESTRICT,
  school_id BIGINT NOT NULL REFERENCES "School"(school_id) ON DELETE RESTRICT,
  
  is_active BOOLEAN NOT NULL DEFAULT TRUE,  -- Only one TRUE per teacher
  
  joined_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  left_at TIMESTAMPTZ DEFAULT NULL,
  leaving_reason TEXT DEFAULT NULL,
  
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  
  -- Consistency: if active, left_at must be NULL; if inactive, left_at must be set
  CONSTRAINT chk_teacher_active_dates
  CHECK (
    (is_active = TRUE AND left_at IS NULL)
    OR
    (is_active = FALSE AND left_at IS NOT NULL)
  )
);

-- Performance indexes
CREATE INDEX idx_teacher_school_enrollment_teacher ON "Teacher_School_Enrollment"(teacher_id);
CREATE INDEX idx_teacher_school_enrollment_school ON "Teacher_School_Enrollment"(school_id);

-- 🚨 KEY RULE: only ONE active enrollment per teacher
CREATE UNIQUE INDEX idx_unique_teacher_active
ON "Teacher_School_Enrollment"(teacher_id)
WHERE is_active = TRUE;

-- Optional: prevent duplicate active rows for same school
CREATE UNIQUE INDEX idx_unique_teacher_school_active
ON "Teacher_School_Enrollment"(teacher_id, school_id)
WHERE is_active = TRUE;





CREATE TABLE "Class_ClassTeacher_Relation" (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  
  class_id BIGINT NOT NULL REFERENCES "Class"(class_id) ON DELETE RESTRICT,
  teacher_id BIGINT NOT NULL REFERENCES "Teacher"(teacher_id) ON DELETE RESTRICT,
  
  created_at TIMESTAMPTZ DEFAULT NOW(),

  -- Prevent duplicate assignment of the SAME teacher to the SAME class
  UNIQUE (class_id, teacher_id)
);




CREATE TABLE "ClassSubject_SubjectTeacher_Relation" (
  link_id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

  class_subject_relation_id BIGINT NOT NULL REFERENCES "Class_Subject_Relation"(class_subject_relation_id) ON DELETE RESTRICT,

  teacher_id BIGINT NOT NULL REFERENCES "Teacher"(teacher_id) ON DELETE RESTRICT,

  created_at TIMESTAMPTZ DEFAULT NOW(),

  -- Prevent duplicate teacher assignment for the same subject entry
  UNIQUE (class_subject_relation_id, teacher_id)
);
